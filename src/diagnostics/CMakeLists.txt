# Also usable standalone: cmake -S src/diagnostics -B build-diagnostics
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(AcidDiagnostics LANGUAGES CXX)
  include(CTest)
endif()
add_library(acid_diagnostics diagnostics.cpp report.cpp)
target_compile_features(acid_diagnostics PUBLIC cxx_std_20)
target_include_directories(acid_diagnostics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(WIN32)
  target_compile_definitions(acid_diagnostics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(acid_diagnostics PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(acid_diagnostics PRIVATE /W4 /EHsc)
endif()
if(BUILD_TESTING)
  add_executable(diagnostics_tests ../../tests/diagnostics_tests.cpp)
  target_link_libraries(diagnostics_tests PRIVATE acid_diagnostics)
  add_test(NAME diagnostics COMMAND diagnostics_tests "${CMAKE_CURRENT_BINARY_DIR}/test-report.json")
endif()
